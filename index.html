✓ Terhubung ke Spreadsheet

URL Google Apps Script Web App sudah tertanam otomatis agar setiap hasil foto & link QR tercatat di Google Spreadsheet dan Google Drive.

Lihat Kode Google Apps Script untuk Spreadsheet & QR (Wajib pakai versi ini) 
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  var folder = DriveApp.getRootFolder();
  
  function saveImg(b64, filename, mime) {
    var blob = Utilities.newBlob(Utilities.base64Decode(b64.split(',')[1]), mime, filename);
    var file = folder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    return file.getId();
  }

  var framedId = saveImg(data.framedBase64, data.sessionId + '-framed.jpg', 'image/jpeg');
  var gifId = data.gifBase64 ? saveImg(data.gifBase64, data.sessionId + '.gif', 'image/gif') : '';
  var takeIds = (data.takesBase64 || []).map(function(t, i) {
    return saveImg(t, data.sessionId + '-take-' + (i+1) + '.jpg', 'image/jpeg');
  });

  var framedUrl = 'https://drive.google.com/uc?export=view&id=' + framedId;
  var gifUrl = gifId ? 'https://drive.google.com/uc?export=view&id=' + gifId : '';
  var takeUrls = takeIds.map(function(id) { return 'https://drive.google.com/uc?export=view&id=' + id; });

  sheet.appendRow([new Date(), data.sessionId, data.layout, data.frameName, framedUrl, gifUrl, JSON.stringify(takeUrls)]);
  return ContentService.createTextOutput(JSON.stringify({status: 'ok', framedUrl: framedUrl, gifUrl: gifUrl, takeUrls: takeUrls})).setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var sessionId = String(e.parameter.session || '').trim();
  var callback = e.parameter.callback;
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var result = {status: 'not_found'};

  for (var i = rows.length - 1; i >= 1; i--) {
    if (String(rows[i][1]).trim() === sessionId) {
      result = {
        status: 'ok',
        sessionId: rows[i][1],
        layout: rows[i][2],
        frameName: rows[i][3],
        framedUrl: rows[i][4],
        gifUrl: rows[i][5],
        takeUrls: JSON.parse(rows[i][6] || '[]')
      };
      break;
    }
  }
  var jsonStr = JSON.stringify(result);
  if (callback) {
    return ContentService.createTextOutput(callback + '(' + jsonStr + ')').setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(jsonStr).setMimeType(ContentService.MimeType.JSON);
}
